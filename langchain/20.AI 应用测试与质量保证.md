```typescript
// src/testing/ai-test-utils.ts
import { ChatOpenAI } from '@langchain/openai';
import { TensorFlowManager } from '@/services/ai/tensorflow-manager';

export interface TestConfig {
  modelName: string;
  testData: any[];
  expectedOutputs?: any[];
  tolerance?: number;
  timeout?: number;
}

export interface TestResult {
  passed: boolean;
  actual: any;
  expected?: any;
  error?: string;
  duration: number;
  metadata?: any;
}

export interface ModelTestSuite {
  name: string;
  tests: ModelTestCase[];
  setup?: () => Promise<void>;
  teardown?: () => Promise<void>;
}

export interface ModelTestCase {
  name: string;
  input: any;
  expected?: any;
  validator?: (actual: any, expected?: any) => boolean;
  timeout?: number;
}

export class AITestUtils {
  private tfManager: TensorFlowManager | null = null;
  private openAIModel: ChatOpenAI | null = null;

  async initialize(): Promise<void> {
    try {
      // 初始化 TensorFlow 管理器
      this.tfManager = new TensorFlowManager();
      await this.tfManager.initialize();

      // 初始化 OpenAI 模型
      this.openAIModel = new ChatOpenAI({
        openAIApiKey: process.env.OPENAI_API_KEY,
        modelName: 'gpt-3.5-turbo',
        temperature: 0.1 // 降低随机性
      });

      console.log('AI 测试工具初始化完成');
    } catch (error) {
      console.error('AI 测试工具初始化失败:', error);
      throw error;
    }
  }

  async testModel(config: TestConfig): Promise<TestResult[]> {
    const results: TestResult[] = [];

    for (let i = 0; i < config.testData.length; i++) {
      const testCase = config.testData[i];
      const expected = config.expectedOutputs?.[i];

      const result = await this.runSingleTest({
        modelName: config.modelName,
        input: testCase,
        expected,
        tolerance: config.tolerance,
        timeout: config.timeout
      });

      results.push(result);
    }

    return results;
  }

  private async runSingleTest(testCase: {
    modelName: string;
    input: any;
    expected?: any;
    tolerance?: number;
    timeout?: number;
  }): Promise<TestResult> {
    const startTime = performance.now();

    try {
      let actual: any;

      // 根据模型类型选择测试方法
      if (testCase.modelName.includes('tensorflow') || testCase.modelName.includes('tf')) {
        actual = await this.testTensorFlowModel(testCase.modelName, testCase.input);
      } else if (testCase.modelName.includes('openai') || testCase.modelName.includes('gpt')) {
        actual = await this.testOpenAIModel(testCase.modelName, testCase.input);
      } else {
        throw new Error(`不支持的模型类型: ${testCase.modelName}`);
      }

      const duration = performance.now() - startTime;
      const passed = this.validateOutput(actual, testCase.expected, testCase.tolerance);

      return {
        passed,
        actual,
        expected: testCase.expected,
        duration,
        metadata: {
          modelName: testCase.modelName,
          inputType: typeof testCase.input
        }
      };
    } catch (error) {
      const duration = performance.now() - startTime;
      return {
        passed: false,
        actual: null,
        expected: testCase.expected,
        error: error.message,
        duration
      };
    }
  }

  private async testTensorFlowModel(modelName: string, input: any): Promise<any> {
    if (!this.tfManager) {
      throw new Error('TensorFlow 管理器未初始化');
    }

    // 将输入转换为张量
    const inputTensor = this.convertToTensor(input);

    // 执行预测
    const result = await this.tfManager.predict(modelName, inputTensor);

    // 清理张量
    inputTensor.dispose();

    return result.predictions;
  }

  private async testOpenAIModel(modelName: string, input: any): Promise<any> {
    if (!this.openAIModel) {
      throw new Error('OpenAI 模型未初始化');
    }

    const response = await this.openAIModel.invoke(input);
    return response.content;
  }

  private convertToTensor(input: any): any {
    if (Array.isArray(input)) {
      return tf.tensor2d([input]);
    } else if (typeof input === 'number') {
      return tf.scalar(input);
    } else {
      throw new Error('不支持的输入类型');
    }
  }

  private validateOutput(actual: any, expected: any, tolerance: number = 0.01): boolean {
    if (expected === undefined) {
      return actual !== null && actual !== undefined;
    }

    if (typeof actual === 'number' && typeof expected === 'number') {
      return Math.abs(actual - expected) <= tolerance;
    }

    if (Array.isArray(actual) && Array.isArray(expected)) {
      if (actual.length !== expected.length) {
        return false;
      }

      for (let i = 0; i < actual.length; i++) {
        if (!this.validateOutput(actual[i], expected[i], tolerance)) {
          return false;
        }
      }
      return true;
    }

    return actual === expected;
  }

  async runTestSuite(suite: ModelTestSuite): Promise<{
    suiteName: string;
    totalTests: number;
    passedTests: number;
    failedTests: number;
    results: TestResult[];
    duration: number;
  }> {
    const startTime = performance.now();
    const results: TestResult[] = [];

    try {
      // 执行设置
      if (suite.setup) {
        await suite.setup();
      }

      // 运行测试用例
      for (const testCase of suite.tests) {
        const result = await this.runTestCase(testCase);
        results.push(result);
      }

      const duration = performance.now() - startTime;
      const passedTests = results.filter(r => r.passed).length;
      const failedTests = results.length - passedTests;

      return {
        suiteName: suite.name,
        totalTests: suite.tests.length,
        passedTests,
        failedTests,
        results,
        duration
      };
    } finally {
      // 执行清理
      if (suite.teardown) {
        await suite.teardown();
      }
    }
  }

  private async runTestCase(testCase: ModelTestCase): Promise<TestResult> {
    const startTime = performance.now();

    try {
      let actual: any;

      // 根据输入类型选择测试方法
      if (typeof testCase.input === 'string') {
        actual = await this.testOpenAIModel('gpt-3.5-turbo', testCase.input);
      } else if (Array.isArray(testCase.input)) {
        actual = await this.testTensorFlowModel('test-model', testCase.input);
      } else {
        throw new Error('不支持的输入类型');
      }

      const duration = performance.now() - startTime;

      // 使用自定义验证器或默认验证
      const passed = testCase.validator
        ? testCase.validator(actual, testCase.expected)
        : this.validateOutput(actual, testCase.expected);

      return {
        passed,
        actual,
        expected: testCase.expected,
        duration
      };
    } catch (error) {
      const duration = performance.now() - startTime;
      return {
        passed: false,
        actual: null,
        expected: testCase.expected,
        error: error.message,
        duration
      };
    }
  }

  generateTestReport(results: TestResult[]): {
    summary: {
      total: number;
      passed: number;
      failed: number;
      passRate: number;
      averageDuration: number;
    };
    details: TestResult[];
    recommendations: string[];
  } {
    const total = results.length;
    const passed = results.filter(r => r.passed).length;
    const failed = total - passed;
    const passRate = total > 0 ? passed / total : 0;
    const averageDuration = results.reduce((sum, r) => sum + r.duration, 0) / total;

    const recommendations: string[] = [];

    if (passRate < 0.8) {
      recommendations.push('测试通过率较低，需要检查模型和测试数据');
    }

    if (averageDuration > 5000) {
      recommendations.push('平均测试时间过长，考虑优化模型性能');
    }

    const failedTests = results.filter(r => !r.passed);
    if (failedTests.length > 0) {
      recommendations.push(`有 ${failedTests.length} 个测试失败，需要修复`);
    }

    return {
      summary: {
        total,
        passed,
        failed,
        passRate,
        averageDuration
      },
      details: results,
      recommendations
    };
  }

  async cleanup(): Promise<void> {
    if (this.tfManager) {
      await this.tfManager.cleanup();
      this.tfManager = null;
    }
    console.log('AI 测试工具已清理');
  }
}
```

### 模型质量评估

```typescript
// src/testing/model-evaluator.ts
export interface EvaluationMetrics {
  accuracy: number;
  precision: number;
  recall: number;
  f1Score: number;
  confusionMatrix: number[][];
  rocAuc?: number;
  prAuc?: number;
}

export interface EvaluationDataset {
  inputs: any[];
  labels: any[];
  metadata?: any[];
}

export interface ModelEvaluationResult {
  metrics: EvaluationMetrics;
  predictions: any[];
  errors: string[];
  duration: number;
  modelVersion: string;
  datasetSize: number;
}

export class ModelEvaluator {
  private testUtils: AITestUtils;

  constructor(testUtils: AITestUtils) {
    this.testUtils = testUtils;
  }

  async evaluateModel(
    modelName: string,
    dataset: EvaluationDataset,
    config: {
      batchSize?: number;
      timeout?: number;
      tolerance?: number;
    } = {}
  ): Promise<ModelEvaluationResult> {
    const startTime = performance.now();
    const predictions: any[] = [];
    const errors: string[] = [];

    try {
      // 批量处理数据集
      const batchSize = config.batchSize || 10;
      const batches = this.createBatches(dataset.inputs, batchSize);

      for (const batch of batches) {
        try {
          const batchPredictions = await this.testUtils.testModel({
            modelName,
            testData: batch,
            timeout: config.timeout
          });

          predictions.push(...batchPredictions.map(r => r.actual));
        } catch (error) {
          errors.push(`批次处理失败: ${error.message}`);
        }
      }

      // 计算评估指标
      const metrics = this.calculateMetrics(predictions, dataset.labels);

      const duration = performance.now() - startTime;

      return {
        metrics,
        predictions,
        errors,
        duration,
        modelVersion: '1.0.0', // 从模型元数据获取
        datasetSize: dataset.inputs.length
      };
    } catch (error) {
      throw new Error(`模型评估失败: ${error.message}`);
    }
  }

  private createBatches<T>(items: T[], batchSize: number): T[][] {
    const batches: T[][] = [];
    for (let i = 0; i < items.length; i += batchSize) {
      batches.push(items.slice(i, i + batchSize));
    }
    return batches;
  }

  private calculateMetrics(predictions: any[], labels: any[]): EvaluationMetrics {
    if (predictions.length !== labels.length) {
      throw new Error('预测结果和标签数量不匹配');
    }

    // 计算混淆矩阵
    const confusionMatrix = this.calculateConfusionMatrix(predictions, labels);

    // 计算各种指标
    const accuracy = this.calculateAccuracy(confusionMatrix);
    const precision = this.calculatePrecision(confusionMatrix);
    const recall = this.calculateRecall(confusionMatrix);
    const f1Score = this.calculateF1Score(precision, recall);

    return {
      accuracy,
      precision,
      recall,
      f1Score,
      confusionMatrix
    };
  }

  private calculateConfusionMatrix(predictions: any[], labels: any[]): number[][] {
    // 获取所有唯一的标签
    const uniqueLabels = [...new Set([...predictions, ...labels])];
    const labelMap = new Map(uniqueLabels.map((label, index) => [label, index]));

    // 初始化混淆矩阵
    const matrix = Array(uniqueLabels.length)
      .fill(null)
      .map(() => Array(uniqueLabels.length).fill(0));

    // 填充混淆矩阵
    for (let i = 0; i < predictions.length; i++) {
      const predIndex = labelMap.get(predictions[i]);
      const labelIndex = labelMap.get(labels[i]);

      if (predIndex !== undefined && labelIndex !== undefined) {
        matrix[labelIndex][predIndex]++;
      }
    }

    return matrix;
  }

  private calculateAccuracy(confusionMatrix: number[][]): number {
    const total = confusionMatrix.reduce((sum, row) =>
      sum + row.reduce((rowSum, cell) => rowSum + cell, 0), 0
    );

    const correct = confusionMatrix.reduce((sum, row, index) =>
      sum + row[index], 0
    );

    return total > 0 ? correct / total : 0;
  }

  private calculatePrecision(confusionMatrix: number[][]): number {
    const precisions: number[] = [];

    for (let i = 0; i < confusionMatrix.length; i++) {
      const columnSum = confusionMatrix.reduce((sum, row) => sum + row[i], 0);
      const precision = columnSum > 0 ? confusionMatrix[i][i] / columnSum : 0;
      precisions.push(precision);
    }

    return precisions.reduce((sum, p) => sum + p, 0) / precisions.length;
  }

  private calculateRecall(confusionMatrix: number[][]): number {
    const recalls: number[] = [];

    for (let i = 0; i < confusionMatrix.length; i++) {
      const rowSum = confusionMatrix[i].reduce((sum, cell) => sum + cell, 0);
      const recall = rowSum > 0 ? confusionMatrix[i][i] / rowSum : 0;
      recalls.push(recall);
    }

    return recalls.reduce((sum, r) => sum + r, 0) / recalls.length;
  }

  private calculateF1Score(precision: number, recall: number): number {
    if (precision + recall === 0) return 0;
    return 2 * (precision * recall) / (precision + recall);
  }

  async compareModels(
    models: string[],
    dataset: EvaluationDataset
  ): Promise<{
    results: Map<string, ModelEvaluationResult>;
    ranking: Array<{ model: string; score: number }>;
    recommendations: string[];
  }> {
    const results = new Map<string, ModelEvaluationResult>();
    const recommendations: string[] = [];

    // 评估每个模型
    for (const modelName of models) {
      try {
        const result = await this.evaluateModel(modelName, dataset);
        results.set(modelName, result);
      } catch (error) {
        console.error(`模型 ${modelName} 评估失败:`, error);
        recommendations.push(`模型 ${modelName} 评估失败: ${error.message}`);
      }
    }

    // 根据 F1 分数排序
    const ranking = Array.from(results.entries())
      .map(([model, result]) => ({
        model,
        score: result.metrics.f1Score
      }))
      .sort((a, b) => b.score - a.score);

    // 生成建议
    if (ranking.length > 0) {
      const bestModel = ranking[0];
      recommendations.push(`最佳模型: ${bestModel.model} (F1: ${bestModel.score.toFixed(3)})`);

      if (ranking.length > 1) {
        const secondBest = ranking[1];
        const improvement = bestModel.score - secondBest.score;
        recommendations.push(`相比第二名模型提升: ${(improvement * 100).toFixed(1)}%`);
      }
    }

    return {
      results,
      ranking,
      recommendations
    };
  }

  generateEvaluationReport(result: ModelEvaluationResult): {
    summary: {
      overallScore: number;
      performance: 'excellent' | 'good' | 'fair' | 'poor';
      recommendations: string[];
    };
    details: {
      metrics: EvaluationMetrics;
      errors: string[];
      duration: number;
    };
  } {
    const overallScore = result.metrics.f1Score;

    let performance: 'excellent' | 'good' | 'fair' | 'poor';
    if (overallScore >= 0.9) {
      performance = 'excellent';
    } else if (overallScore >= 0.8) {
      performance = 'good';
    } else if (overallScore >= 0.6) {
      performance = 'fair';
    } else {
      performance = 'poor';
    }

    const recommendations: string[] = [];

    if (result.metrics.accuracy < 0.8) {
      recommendations.push('准确率较低，建议检查训练数据和模型架构');
    }

    if (result.metrics.precision < 0.7) {
      recommendations.push('精确率较低，建议调整分类阈值');
    }

    if (result.metrics.recall < 0.7) {
      recommendations.push('召回率较低，建议增加训练数据');
    }

    if (result.errors.length > 0) {
      recommendations.push(`有 ${result.errors.length} 个错误需要修复`);
    }

    return {
      summary: {
        overallScore,
        performance,
        recommendations
      },
      details: {
        metrics: result.metrics,
        errors: result.errors,
        duration: result.duration
      }
    };
  }
}
```

```typescript
describe('TensorFlowManager', () => {
  let tfManager: TensorFlowManager;
  let testUtils: AITestUtils;

  beforeAll(async () => {
    testUtils = new AITestUtils();
    await testUtils.initialize();

    tfManager = new TensorFlowManager();

        outputShape: [1, 2],

      expect(results).toHaveLength(2);

      const result = await tfManager.predict('test-model', input);
      const endTime = performance.now();

      const totalTime = endTime - startTime;
      expect(totalTime).toBeLessThan(1000); // 应该在 1 秒内完成

      input.dispose();
```

```typescript
  duration: number;
  error?: string;
  metadata?: any;

  lines: number;
  files: Array<{
    path: string;

  p95ResponseTime: number;
  p99ResponseTime: number;
  throughput: number;
  memoryUsage: number;
  cpuUsage: number;

export class TestReportGenerator {
  async generateReport(testResults: any[]): Promise<TestReport> {
    const summary = this.calculateSummary(testResults);
    const suites = this.groupBySuite(testResults);
    const coverage = await this.generateCoverageReport();
    const performance = await this.generatePerformanceReport();
    const recommendations = this.generateRecommendations(summary, coverage, performance);

    return {
      summary,
      suites,
      coverage,
      performance,
      recommendations,
      timestamp: new Date().toISOString()
    };
  }

  private calculateSummary(testResults: any[]): TestReport['summary'] {
    const totalTests = testResults.length;
    const passedTests = testResults.filter(r => r.status === 'passed').length;
    const failedTests = testResults.filter(r => r.status === 'failed').length;
    const skippedTests = testResults.filter(r => r.status === 'skipped').length;
    const duration = testResults.reduce((sum, r) => sum + r.duration, 0);
    const passRate = totalTests > 0 ? passedTests / totalTests : 0;

    return {
      totalTests,
      passedTests,
      failedTests,
      skippedTests,
      duration,
      passRate
    };
  }

  private groupBySuite(testResults: any[]): TestSuiteReport[] {
    const suiteMap = new Map<string, TestCaseReport[]>();

    testResults.forEach(result => {
      const suiteName = result.suite || 'default';
      if (!suiteMap.has(suiteName)) {
        suiteMap.set(suiteName, []);
      }

      suiteMap.get(suiteName)!.push({
        name: result.name,
        status: result.status,
        duration: result.duration,
        error: result.error,
        metadata: result.metadata
      });
    });

    return Array.from(suiteMap.entries()).map(([name, tests]) => ({
      name,
      tests,
      summary: {
        total: tests.length,
        passed: tests.filter(t => t.status === 'passed').length,
        failed: tests.filter(t => t.status === 'failed').length,
        duration: tests.reduce((sum, t) => sum + t.duration, 0)
      }
    }));

    return {

    if (performance.averageResponseTime > 500) {

    }

    if (performance.memoryUsage > 200) {

    }

    if (summary.failedTests > 0) {
      recommendations.push(`有 ${summary.failedTests} 个测试失败，需要优先修复`);

                <div class="value">${performance.memoryUsage.toFixed(1)} MB</div>
            </div>
        </div>
    </div>

    ${report.suites.map(suite => `
        <div class="suite">
            <h2>${suite.name}</h2>
            <p>总计: ${suite.summary.total} | 通过: ${suite.summary.passed} | 失败: ${suite.summary.failed} | 耗时: ${suite.summary.duration.toFixed(2)}ms</p>
            ${suite.tests.map(test => `
                <div class="test-case ${test.status}">
                    <strong>${test.name}</strong> - ${test.status} (${test.duration.toFixed(2)}ms)
                    ${test.error ? `<br><small>错误: ${test.error}</small>` : ''}
                </div>
            `).join('')}
        </div>
    `).join('')}

    <div class="recommendations">
        <h2>改进建议</h2>
        <ul>
            ${report.recommendations.map(rec => `<li>${rec}</li>`).join('')}
        </ul>
    </div>
</body>
</html>`;

    return html;
  }

  async saveReport(report: TestReport, outputPath: string): Promise<void> {
    const fs = await import('fs/promises');
    const path = await import('path');

    // 确保输出目录存在
    const outputDir = path.dirname(outputPath);
    await fs.mkdir(outputDir, { recursive: true });

    // 保存 JSON 报告
    await fs.writeFile(outputPath, JSON.stringify(report, null, 2));

    // 生成并保存 HTML 报告
    const htmlReport = await this.generateHTMLReport(report);
    const htmlPath = outputPath.replace('.json', '.html');
    await fs.writeFile(htmlPath, htmlReport);

    console.log(`测试报告已保存到: ${outputPath}`);
    console.log(`HTML 报告已保存到: ${htmlPath}`);
  }
}
```
