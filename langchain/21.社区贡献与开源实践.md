### 项目发现与评估

```typescript
// src/contributing/project-evaluator.ts
export interface ProjectInfo {
  name: string;
  description: string;
  repository: string;
  stars: number;
  forks: number;
  issues: number;
  pullRequests: number;
  lastCommit: Date;
  license: string;
  language: string;
  topics: string[];
  contributors: number;
  healthScore: number;
}

export interface ContributionOpportunity {
  type: 'bug' | 'feature' | 'documentation' | 'test' | 'refactor';
  title: string;
  description: string;
  difficulty: 'beginner' | 'intermediate' | 'advanced';
  labels: string[];
  url: string;
  createdAt: Date;
  comments: number;
}

export class ProjectEvaluator {
  async evaluateProject(repository: string): Promise<ProjectInfo> {
    try {
      // 获取仓库基本信息
      const repoInfo = await this.fetchRepositoryInfo(repository);

      // 获取贡献者信息
      const contributors = await this.fetchContributors(repository);

      // 获取问题信息
      const issues = await this.fetchIssues(repository);

      // 计算健康分数
      const healthScore = this.calculateHealthScore(repoInfo, contributors, issues);

      return {
        name: repoInfo.name,
        description: repoInfo.description,
        repository: repository,
        stars: repoInfo.stargazers_count,
        forks: repoInfo.forks_count,
        issues: repoInfo.open_issues_count,
        pullRequests: await this.fetchPullRequestsCount(repository),
        lastCommit: new Date(repoInfo.updated_at),
        license: repoInfo.license?.name || 'Unknown',
        language: repoInfo.language || 'Unknown',
        topics: repoInfo.topics || [],
        contributors: contributors.length,
        healthScore
      };
    } catch (error) {
      console.error('项目评估失败:', error);
      throw error;
    }
  }

  private async fetchRepositoryInfo(repository: string): Promise<any> {
    const [owner, repo] = repository.split('/');
    const response = await fetch(`https://api.github.com/repos/${owner}/${repo}`);

    if (!response.ok) {
      throw new Error(`无法获取仓库信息: ${response.statusText}`);
    }

    return await response.json();
  }

  private async fetchContributors(repository: string): Promise<any[]> {
    const [owner, repo] = repository.split('/');
    const response = await fetch(`https://api.github.com/repos/${owner}/${repo}/contributors`);

    if (!response.ok) {
      return [];
    }

    return await response.json();
  }

  private async fetchIssues(repository: string): Promise<any[]> {
    const [owner, repo] = repository.split('/');
    const response = await fetch(`https://api.github.com/repos/${owner}/${repo}/issues?state=open`);

    if (!response.ok) {
      return [];
    }

    return await response.json();
  }

  private async fetchPullRequestsCount(repository: string): Promise<number> {
    const [owner, repo] = repository.split('/');
    const response = await fetch(`https://api.github.com/repos/${owner}/${repo}/pulls?state=open`);

    if (!response.ok) {
      return 0;
    }

    const pulls = await response.json();
    return pulls.length;
  }

  private calculateHealthScore(repoInfo: any, contributors: any[], issues: any[]): number {
    let score = 0;

    // 基于星标数量 (0-20分)
    score += Math.min(20, Math.log10(repoInfo.stargazers_count + 1) * 5);

    // 基于贡献者数量 (0-20分)
    score += Math.min(20, contributors.length * 2);

    // 基于最近活动 (0-20分)
    const daysSinceUpdate = (Date.now() - new Date(repoInfo.updated_at).getTime()) / (1000 * 60 * 60 * 24);
    score += Math.max(0, 20 - daysSinceUpdate);

    // 基于问题响应 (0-20分)
    const recentIssues = issues.filter(issue => {
      const daysSinceCreated = (Date.now() - new Date(issue.created_at).getTime()) / (1000 * 60 * 60 * 24);
      return daysSinceCreated < 30;
    });
    score += Math.min(20, recentIssues.length * 2);

    // 基于许可证 (0-10分)
    if (repoInfo.license) {
      score += 10;
    }

    // 基于 README 质量 (0-10分)
    if (repoInfo.description && repoInfo.description.length > 50) {
      score += 10;
    }

    return Math.min(100, score);
  }

  async findContributionOpportunities(repository: string): Promise<ContributionOpportunity[]> {
    const [owner, repo] = repository.split('/');
    const opportunities: ContributionOpportunity[] = [];

    try {
      // 获取标记为 "good first issue" 的问题
      const goodFirstIssues = await this.fetchIssuesByLabel(repository, 'good first issue');
      opportunities.push(...goodFirstIssues.map(issue => this.mapIssueToOpportunity(issue, 'beginner')));

      // 获取标记为 "help wanted" 的问题
      const helpWantedIssues = await this.fetchIssuesByLabel(repository, 'help wanted');
      opportunities.push(...helpWantedIssues.map(issue => this.mapIssueToOpportunity(issue, 'intermediate')));

      // 获取文档相关的问题
      const docIssues = await this.fetchIssuesByLabel(repository, 'documentation');
      opportunities.push(...docIssues.map(issue => this.mapIssueToOpportunity(issue, 'beginner', 'documentation')));

      // 获取测试相关的问题
      const testIssues = await this.fetchIssuesByLabel(repository, 'test');
      opportunities.push(...testIssues.map(issue => this.mapIssueToOpportunity(issue, 'intermediate', 'test')));

    } catch (error) {
      console.error('获取贡献机会失败:', error);
    }

    return opportunities.sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime());
  }

  private async fetchIssuesByLabel(repository: string, label: string): Promise<any[]> {
    const [owner, repo] = repository.split('/');
    const response = await fetch(`https://api.github.com/repos/${owner}/${repo}/issues?labels=${label}&state=open`);

    if (!response.ok) {
      return [];
    }

    return await response.json();
  }

  private mapIssueToOpportunity(issue: any, difficulty: 'beginner' | 'intermediate' | 'advanced', type?: string): ContributionOpportunity {
    return {
      type: type as any || this.determineIssueType(issue),
      title: issue.title,
      description: issue.body,
      difficulty,
      labels: issue.labels.map((label: any) => label.name),
      url: issue.html_url,
      createdAt: new Date(issue.created_at),
      comments: issue.comments
    };
  }

  private determineIssueType(issue: any): 'bug' | 'feature' | 'documentation' | 'test' | 'refactor' {
    const title = issue.title.toLowerCase();
    const body = issue.body?.toLowerCase() || '';
    const labels = issue.labels.map((label: any) => label.name.toLowerCase());

    if (labels.includes('bug') || title.includes('bug') || body.includes('error')) {
      return 'bug';
    }

    if (labels.includes('feature') || title.includes('feature') || title.includes('enhancement')) {
      return 'feature';
    }

    if (labels.includes('documentation') || title.includes('doc') || title.includes('readme')) {
      return 'documentation';
    }

    if (labels.includes('test') || title.includes('test') || title.includes('testing')) {
      return 'test';
    }

    if (labels.includes('refactor') || title.includes('refactor') || title.includes('cleanup')) {
      return 'refactor';
    }

    return 'feature'; // 默认为功能请求
  }

  generateProjectReport(projectInfo: ProjectInfo, opportunities: ContributionOpportunity[]): string {
    return `
# 项目评估报告: ${projectInfo.name}

## 基本信息
- **仓库**: ${projectInfo.repository}
- **描述**: ${projectInfo.description}
- **语言**: ${projectInfo.language}
- **许可证**: ${projectInfo.license}
- **健康分数**: ${projectInfo.healthScore}/100

## 统计数据
- ⭐ 星标: ${projectInfo.stars}
- 🍴 Fork: ${projectInfo.forks}
- 🐛 开放问题: ${projectInfo.issues}
- 🔄 开放 PR: ${projectInfo.pullRequests}
- 👥 贡献者: ${projectInfo.contributors}
- 📅 最后更新: ${projectInfo.lastCommit.toLocaleDateString()}

## 贡献机会 (${opportunities.length} 个)
${opportunities.slice(0, 10).map(opp => `
### ${opp.title}
- **类型**: ${opp.type}
- **难度**: ${opp.difficulty}
- **标签**: ${opp.labels.join(', ')}
- **创建时间**: ${opp.createdAt.toLocaleDateString()}
- **链接**: ${opp.url}
`).join('')}

## 建议
${this.generateRecommendations(projectInfo, opportunities)}
`;
  }

  private generateRecommendations(projectInfo: ProjectInfo, opportunities: ContributionOpportunity[]): string {
    const recommendations: string[] = [];

    if (projectInfo.healthScore < 50) {
      recommendations.push('项目健康分数较低，建议谨慎贡献');
    }

    if (projectInfo.issues > 100) {
      recommendations.push('开放问题较多，建议优先处理 Bug 修复');
    }

    const beginnerOpportunities = opportunities.filter(opp => opp.difficulty === 'beginner');
    if (beginnerOpportunities.length > 0) {
      recommendations.push(`有 ${beginnerOpportunities.length} 个适合初学者的贡献机会`);
    }

    if (projectInfo.lastCommit < new Date(Date.now() - 30 * 24 * 60 * 60 * 1000)) {
      recommendations.push('项目最近活动较少，建议联系维护者确认项目状态');
    }

    return recommendations.join('\n- ');
  }
}
```

### 贡献流程管理

```typescript
// src/contributing/contribution-manager.ts
export interface ContributionPlan {
  id: string;
  repository: string;
  opportunity: ContributionOpportunity;
  status: 'planning' | 'in_progress' | 'review' | 'merged' | 'closed';
  startDate: Date;
  targetDate: Date;
  actualDate?: Date;
  notes: string[];
  progress: number;
}

export interface ContributionChecklist {
  id: string;
  task: string;
  completed: boolean;
  completedAt?: Date;
  notes?: string;
}

export class ContributionManager {
  private contributions: Map<string, ContributionPlan> = new Map();
  private checklists: Map<string, ContributionChecklist[]> = new Map();

  async startContribution(opportunity: ContributionOpportunity): Promise<ContributionPlan> {
    const plan: ContributionPlan = {
      id: this.generateId(),
      repository: this.extractRepositoryFromUrl(opportunity.url),
      opportunity,
      status: 'planning',
      startDate: new Date(),
      targetDate: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000), // 默认 7 天
      notes: [],
      progress: 0
    };

    this.contributions.set(plan.id, plan);
    await this.createChecklist(plan.id, opportunity);

    return plan;
  }

  private extractRepositoryFromUrl(url: string): string {
    const match = url.match(/github\.com\/([^\/]+)\/([^\/]+)/);
    if (match) {
      return `${match[1]}/${match[2]}`;
    }
    throw new Error('无法从 URL 提取仓库信息');
  }

  private generateId(): string {
    return `contrib_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }

  private async createChecklist(planId: string, opportunity: ContributionOpportunity): Promise<void> {
    const checklist: ContributionChecklist[] = [
      {
        id: 'fork_repo',
        task: 'Fork 仓库到个人账户',
        completed: false
      },
      {
        id: 'clone_repo',
        task: '克隆仓库到本地',
        completed: false
      },
      {
        id: 'create_branch',
        task: '创建功能分支',
        completed: false
      },
      {
        id: 'setup_env',
        task: '设置开发环境',
        completed: false
      },
      {
        id: 'understand_code',
        task: '理解相关代码',
        completed: false
      },
      {
        id: 'implement_solution',
        task: '实现解决方案',
        completed: false
      },
      {
        id: 'write_tests',
        task: '编写测试用例',
        completed: false
      },
      {
        id: 'update_docs',
        task: '更新文档',
        completed: false
      },
      {
        id: 'run_tests',
        task: '运行测试',
        completed: false
      },
      {
        id: 'create_pr',
        task: '创建 Pull Request',
        completed: false
      }
    ];

    this.checklists.set(planId, checklist);
  }

  async updateContributionStatus(planId: string, status: ContributionPlan['status']): Promise<void> {
    const plan = this.contributions.get(planId);
    if (!plan) {
      throw new Error('贡献计划不存在');
    }

    plan.status = status;

    if (status === 'merged' || status === 'closed') {
      plan.actualDate = new Date();
    }

    await this.updateProgress(planId);
  }

  async completeChecklistItem(planId: string, itemId: string, notes?: string): Promise<void> {
    const checklist = this.checklists.get(planId);
    if (!checklist) {
      throw new Error('检查清单不存在');
    }

    const item = checklist.find(item => item.id === itemId);
    if (!item) {
      throw new Error('检查清单项不存在');
    }

    item.completed = true;
    item.completedAt = new Date();
    if (notes) {
      item.notes = notes;
    }

    await this.updateProgress(planId);
  }

  private async updateProgress(planId: string): Promise<void> {
    const plan = this.contributions.get(planId);
    const checklist = this.checklists.get(planId);

    if (!plan || !checklist) {
      return;
    }

    const completedItems = checklist.filter(item => item.completed).length;
    plan.progress = (completedItems / checklist.length) * 100;

    // 根据进度更新状态
    if (plan.progress >= 100 && plan.status === 'in_progress') {
      plan.status = 'review';
    } else if (plan.progress > 0 && plan.status === 'planning') {
      plan.status = 'in_progress';
    }
  }

  async addNote(planId: string, note: string): Promise<void> {
    const plan = this.contributions.get(planId);
    if (!plan) {
      throw new Error('贡献计划不存在');
    }

    plan.notes.push(`[${new Date().toISOString()}] ${note}`);
  }

  getContribution(planId: string): ContributionPlan | undefined {
    return this.contributions.get(planId);
  }

  getChecklist(planId: string): ContributionChecklist[] | undefined {
    return this.checklists.get(planId);
  }

  getAllContributions(): ContributionPlan[] {
    return Array.from(this.contributions.values());
  }

  getContributionsByStatus(status: ContributionPlan['status']): ContributionPlan[] {
    return Array.from(this.contributions.values()).filter(plan => plan.status === status);
  }

  async generateContributionReport(planId: string): Promise<string> {
    const plan = this.contributions.get(planId);
    const checklist = this.checklists.get(planId);

    if (!plan || !checklist) {
      throw new Error('贡献计划不存在');
    }

    const completedItems = checklist.filter(item => item.completed);
    const remainingItems = checklist.filter(item => !item.completed);

    return `
# 贡献进度报告

## 基本信息
- **贡献 ID**: ${plan.id}
- **仓库**: ${plan.repository}
- **问题**: ${plan.opportunity.title}
- **状态**: ${plan.status}
- **进度**: ${plan.progress.toFixed(1)}%

## 时间线
- **开始时间**: ${plan.startDate.toLocaleString()}
- **目标完成**: ${plan.targetDate.toLocaleString()}
- **实际完成**: ${plan.actualDate?.toLocaleString() || '未完成'}

## 检查清单进度
### 已完成 (${completedItems.length}/${checklist.length})
${completedItems.map(item => `
- ✅ ${item.task} (${item.completedAt?.toLocaleString()})
  ${item.notes ? `  - 备注: ${item.notes}` : ''}
`).join('')}

### 待完成 (${remainingItems.length})
${remainingItems.map(item => `
- ⏳ ${item.task}
`).join('')}

## 备注
${plan.notes.map(note => `- ${note}`).join('\n')}

## 下一步行动
${this.generateNextSteps(plan, checklist)}
`;
  }

  private generateNextSteps(plan: ContributionPlan, checklist: ContributionChecklist[]): string {
    const nextItem = checklist.find(item => !item.completed);

    if (!nextItem) {
      return '所有任务已完成，等待代码审查';
    }

    const suggestions: string[] = [];

    switch (nextItem.id) {
      case 'fork_repo':
        suggestions.push('访问 GitHub 仓库页面，点击 Fork 按钮');
        break;
      case 'clone_repo':
        suggestions.push('使用 git clone 命令克隆仓库到本地');
        break;
      case 'create_branch':
        suggestions.push('创建新的功能分支，命名规范: feature/issue-number-description');
        break;
      case 'setup_env':
        suggestions.push('安装依赖，配置开发环境');
        break;
      case 'understand_code':
        suggestions.push('阅读相关代码，理解项目结构和实现逻辑');
        break;
      case 'implement_solution':
        suggestions.push('根据问题描述实现解决方案');
        break;
      case 'write_tests':
        suggestions.push('为新增功能编写测试用例');
        break;
      case 'update_docs':
        suggestions.push('更新相关文档和注释');
        break;
      case 'run_tests':
        suggestions.push('运行所有测试，确保没有破坏现有功能');
        break;
      case 'create_pr':
        suggestions.push('推送代码并创建 Pull Request');
        break;
    }

    return suggestions.join('\n- ');
  }
}
```

```typescript
// src/writing/blog-manager.ts
export interface BlogPost {
  id: string;
  title: string;
  content: string;
  excerpt: string;
  tags: string[];
  category: string;
  status: 'draft' | 'published' | 'archived';
  createdAt: Date;
  updatedAt: Date;
  publishedAt?: Date;
  readTime: number;
  wordCount: number;
  views: number;
  likes: number;
  comments: number;
}

export interface WritingGoal {
  id: string;
  title: string;
  description: string;
  targetPosts: number;
  currentPosts: number;
  targetWords: number;
  currentWords: number;
  deadline: Date;
  status: 'active' | 'completed' | 'paused';
}

export class BlogManager {
  private posts: Map<string, BlogPost> = new Map();
  private goals: Map<string, WritingGoal> = new Map();

  async createPost(postData: {
    title: string;
    content: string;
    tags: string[];
    category: string;
  }): Promise<BlogPost> {
    const post: BlogPost = {
      id: this.generateId(),
      title: postData.title,
      content: postData.content,
      excerpt: this.generateExcerpt(postData.content),
      tags: postData.tags,
      category: postData.category,
      status: 'draft',
      createdAt: new Date(),
      updatedAt: new Date(),
      readTime: this.calculateReadTime(postData.content),
      wordCount: this.countWords(postData.content),
      views: 0,
      likes: 0,
      comments: 0
    };

    this.posts.set(post.id, post);
    return post;
  }

  private generateId(): string {
    return `post_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }

  private generateExcerpt(content: string, maxLength: number = 200): string {
    // 移除 Markdown 标记
    const plainText = content.replace(/[#*`\[\]()]/g, '').replace(/\n/g, ' ');

    if (plainText.length <= maxLength) {
      return plainText;
    }

    return plainText.substring(0, maxLength).trim() + '...';
  }

  private calculateReadTime(content: string): number {
    const wordsPerMinute = 200; // 平均阅读速度
    const wordCount = this.countWords(content);
    return Math.ceil(wordCount / wordsPerMinute);
  }

  private countWords(content: string): number {
    return content.split(/\s+/).filter(word => word.length > 0).length;
  }

  async updatePost(postId: string, updates: Partial<BlogPost>): Promise<void> {
    const post = this.posts.get(postId);
    if (!post) {
      throw new Error('文章不存在');
    }

    Object.assign(post, updates);
    post.updatedAt = new Date();

    // 如果内容更新，重新计算相关指标
    if (updates.content) {
      post.excerpt = this.generateExcerpt(updates.content);
      post.readTime = this.calculateReadTime(updates.content);
      post.wordCount = this.countWords(updates.content);
    }
  }

  async publishPost(postId: string): Promise<void> {
    const post = this.posts.get(postId);
    if (!post) {
      throw new Error('文章不存在');
    }

    post.status = 'published';
    post.publishedAt = new Date();
    post.updatedAt = new Date();

    // 更新写作目标进度
    await this.updateWritingGoals(post);
  }

  async createWritingGoal(goalData: {
    title: string;
    description: string;
    targetPosts: number;
    targetWords: number;
    deadline: Date;
  }): Promise<WritingGoal> {
    const goal: WritingGoal = {
      id: this.generateId(),
      title: goalData.title,
      description: goalData.description,
      targetPosts: goalData.targetPosts,
      currentPosts: 0,
      targetWords: goalData.targetWords,
      currentWords: 0,
      deadline: goalData.deadline,
      status: 'active'
    };

    this.goals.set(goal.id, goal);
    return goal;
  }

  private async updateWritingGoals(post: BlogPost): Promise<void> {
    for (const goal of this.goals.values()) {
      if (goal.status === 'active') {
        goal.currentPosts++;
        goal.currentWords += post.wordCount;

        // 检查是否完成目标
        if (goal.currentPosts >= goal.targetPosts && goal.currentWords >= goal.targetWords) {
          goal.status = 'completed';
        }
      }
    }
  }

  getPost(postId: string): BlogPost | undefined {
    return this.posts.get(postId);
  }

  getAllPosts(): BlogPost[] {
    return Array.from(this.posts.values());
  }

  getPublishedPosts(): BlogPost[] {
    return Array.from(this.posts.values()).filter(post => post.status === 'published');
  }

  getPostsByCategory(category: string): BlogPost[] {
    return Array.from(this.posts.values()).filter(post => post.category === category);
  }

  getPostsByTag(tag: string): BlogPost[] {
    return Array.from(this.posts.values()).filter(post => post.tags.includes(tag));
  }

  searchPosts(query: string): BlogPost[] {
    const lowercaseQuery = query.toLowerCase();
    return Array.from(this.posts.values()).filter(post =>
      post.title.toLowerCase().includes(lowercaseQuery) ||
      post.content.toLowerCase().includes(lowercaseQuery) ||
      post.tags.some(tag => tag.toLowerCase().includes(lowercaseQuery))
    );
  }

  getWritingGoals(): WritingGoal[] {
    return Array.from(this.goals.values());
  }

  getActiveGoals(): WritingGoal[] {
    return Array.from(this.goals.values()).filter(goal => goal.status === 'active');
  }

  async generateWritingReport(): Promise<{
    summary: {
      totalPosts: number;
      publishedPosts: number;
      totalWords: number;
      averageReadTime: number;
      totalViews: number;
      totalLikes: number;
    };
    categories: Array<{ name: string; count: number; words: number }>;
    tags: Array<{ name: string; count: number }>;
    goals: WritingGoal[];
    recommendations: string[];
  }> {
    const posts = this.getAllPosts();
    const publishedPosts = this.getPublishedPosts();

    const totalWords = posts.reduce((sum, post) => sum + post.wordCount, 0);
    const averageReadTime = posts.length > 0 ? posts.reduce((sum, post) => sum + post.readTime, 0) / posts.length : 0;
    const totalViews = posts.reduce((sum, post) => sum + post.views, 0);
    const totalLikes = posts.reduce((sum, post) => sum + post.likes, 0);

    // 统计分类
    const categoryMap = new Map<string, { count: number; words: number }>();
    posts.forEach(post => {
      const existing = categoryMap.get(post.category) || { count: 0, words: 0 };
      existing.count++;
      existing.words += post.wordCount;
      categoryMap.set(post.category, existing);
    });

    const categories = Array.from(categoryMap.entries()).map(([name, data]) => ({
      name,
      count: data.count,
      words: data.words
    }));

    // 统计标签
    const tagMap = new Map<string, number>();
    posts.forEach(post => {
      post.tags.forEach(tag => {
        tagMap.set(tag, (tagMap.get(tag) || 0) + 1);
      });
    });

    const tags = Array.from(tagMap.entries()).map(([name, count]) => ({ name, count }));

    // 生成建议
    const recommendations: string[] = [];

    if (publishedPosts.length < 5) {
      recommendations.push('建议增加发布频率，保持内容更新');
    }

    if (averageReadTime > 10) {
      recommendations.push('文章阅读时间较长，考虑拆分或简化内容');
    }

    if (totalViews < 100) {
      recommendations.push('文章浏览量较低，建议优化标题和推广策略');
    }

    const activeGoals = this.getActiveGoals();
    if (activeGoals.length === 0) {
      recommendations.push('建议设置新的写作目标，保持持续输出');
    }

    return {
      summary: {
        totalPosts: posts.length,
        publishedPosts: publishedPosts.length,
        totalWords,
        averageReadTime: Math.round(averageReadTime),
        totalViews,
        totalLikes
      },
      categories,
      tags,
      goals: this.getWritingGoals(),
      recommendations
    };
  }
}
```
